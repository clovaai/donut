synthtiger==1.2.1

# synthtiger 1.2.1 gọi ImageFont.getsize(), API này bị xoá ở Pillow 10.

# nên chương trình treo im lặng (chạy với -v mới thấy traceback).

# imgaug (synthtiger phụ thuộc) dùng np.sctypes, bị xoá ở NumPy 2.
numpy<2

# opencv-python>=5 yêu cầu numpy>=2, xung đột với dòng trên.
opencv-python<5

# dùng cho tools/check_fonts.py (synthtiger cũng đã kéo về sẵn)
fonttools>=4.0
